library(shiny)

ui <- fluidPage(
  h2("Give an example of a function."),

  textInput(
    inputId = "function_answer",
    label = "Type your example here:"
  ),

  h3("You typed:"),

  textOutput("typed_answer")
)

server <- function(input, output) {

  output$typed_answer <- renderText({
    input$function_answer
  })

}

shinyApp(ui = ui, server = server)
