library(shiny)

ui <- fluidPage(
    h2("Give an example of a function."),
    
    textInput(
        inputId = "function_answer",
        label = "Give an example of a function, then click Add Function:"
    ),
    
    actionButton(
        inputId = "add_function",
        label = "Add function"
    ),
    
    h3("Functions entered:"),
    
    uiOutput("function_list")
)

server <- function(input, output, session) {
    
    # Empty character vector to store answers
    functions_entered <- reactiveVal(character())
    
    # Add the current text-box when button is pressed
    observeEvent(input$add_function, {
        
        # Remove extra spaces
        new_function <- trimws(input$function_answer)
        
        # Only add non-empty responses
        if (new_function != "") {
            
            # Add the new response to the list
            functions_entered(c(functions_entered(), new_function))
            
            # Clear the text box
            updateTextInput(session, "function_answer", value = "")
        }
    })
    
    # Display responses as a list
    output$function_list <- renderUI({
        tags$ul(
            lapply(functions_entered(), tags$li)
        )
    })
}

shinyApp(ui = ui, server = server)
